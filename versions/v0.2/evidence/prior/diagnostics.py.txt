def case_dir(name, text=None):
    p=OUT/name
    p.mkdir(exist_ok=False)
    (p/'experiment.py').write_bytes(SOURCE if text is None else text.encode('utf-8'))
    (p/'test_experiment.py').write_bytes(TESTS)

def test_case(path):
    proc=subprocess.run([sys.executable,'-B','-m','unittest','-v'],cwd=path,capture_output=True,text=True,timeout=30)
    transcript=proc.stdout+proc.stderr
    (path/'test-output.txt').write_text(transcript,encoding='utf-8')
    return {'exit_code':proc.returncode,'ran_eight_tests':bool(re.search(r'Ran 8 tests in ',transcript)),
            'suite_passed':proc.returncode==0,'failure_lines':[line for line in transcript.splitlines() if line.startswith(('FAIL:','ERROR:'))]}
